package umu.pds.duolingoBaratero.services;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

import java.util.*;
import javax.swing.JPanel;

import umu.pds.duolingoBaratero.models.*;
import umu.pds.duolingoBaratero.repositories.RepositorioCurso;

public class ServicioPreguntaTest {

    private ServicioPregunta sirve;

    @BeforeEach
    void prepararTodo() {
        sirve = new ServicioPregunta();
    }

    @Test
    void generaPanelitosVacios() {
        JPanel[] panls = sirve.generarLeccion(123);
        assertNotNull(panls);
        assertEquals(0, panls.length);
    }

    @Test
    void sacaPregsBloque() {
        CursoEnProgreso cursin = new CursoEnProgreso();
        Set<Pregunta> pregs = new HashSet<>();
        Pregunta una = new Pregunta("algo", "respuestilla");
        pregs.add(una);
        cursin.setPreguntasBloqueContenido(pregs);

        Set<Pregunta> rs = sirve.obtenerPreguntasDelBloque(cursin);
        assertEquals(1, rs.size());
        assertTrue(rs.contains(una));
    }

    @Test
    void compruebaBienSiEsCorrecta() {
        Pregunta pq = new Pregunta("Capital?", "Paris");
        pq.setOpciones(List.of("Paris", "Madrid", "Roma"));
        assertTrue(sirve.procesarRespuesta(pq, "Paris"));
    }

    @Test
    void compruebaMalSiEsOtra() {
        Pregunta pq = new Pregunta("Capital?", "Paris");
        pq.setOpciones(List.of("Paris", "Madrid", "Roma"));
        assertFalse(sirve.procesarRespuesta(pq, "Madrid"));
    }

    @Test
    void cuantasHayEnBloqueDeContenido() {
        long bloqueId = 321L;
        BloqueContenido bloq = new BloqueContenido();
        bloq.setId(bloqueId);
        bloq.setPreguntas(List.of(
            new Pregunta("Q1", "R1"),
            new Pregunta("Q2", "R2"),
            new Pregunta("Q3", "R3")
        ));
        RepositorioCurso.INSTANCE.agregarBloqueContenido(bloq);

        int total = sirve.obtenerNumPreguntas(bloqueId);
        assertEquals(3, total);
    }

    @Test
    void guardaPregsSinPetar() {
        List<Pregunta> ps = List.of(
            new Pregunta("1+1?", "2"),
            new Pregunta("2+2?", "4")
        );
        CursoPlantilla cursete = new CursoPlantilla("Mates", "Ana", "sencillo", "sumillas");
        assertDoesNotThrow(() -> sirve.guardarPreguntas(ps, cursete));
    }
}
